# R04 v2：DDL、序列和 replica identity 的限制

这章在同样的逻辑订阅上实验边界，而不是凭口号总结。ALTER TABLE 在 primary 上不会自动出现在 subscriber；手动添加 note 后数据传播正常。identity sequence 不是逻辑发布的同步对象：publisher 已经生成 id=2，而 subscriber 仍有自己的序列 last_value=1，本地插入会和已有 id 冲突。

UPDATE/DELETE 需要 replica identity。把 orders 设为 REPLICA IDENTITY NOTHING 后，更新会被拒绝；FULL 提供整行身份后再更新/删除，subscriber 才能跟上。实际表可以使用主键或明确的 REPLICA IDENTITY USING INDEX，FULL 只是本实验的可观察 fallback，可能增加 WAL。

from __future__ import annotations

import time
from collections.abc import Callable

import psycopg


def connect(node: str) -> psycopg.Connection:
    ports = {"primary": 55440, "standby": 55441, "subscriber": 55442}
    return psycopg.connect(host="127.0.0.1", port=ports[node], dbname="r04", user="r04", password="r04_local_only", autocommit=True, connect_timeout=3)


def wait_for(check: Callable[[], bool], label: str, seconds: float = 60) -> None:
    expires = time.monotonic() + seconds
    while time.monotonic() < expires:
        if check():
            return
        time.sleep(0.05)
    raise TimeoutError(label)


def scalar(conn: psycopg.Connection, query: str) -> object:
    row = conn.execute(query).fetchone()
    return None if row is None else row[0]
```

from __future__ import annotations

import json

from common import connect, scalar, wait_for


def setup() -> None:
    with connect("subscriber") as subscriber, connect("primary") as primary:
        subscriber.execute("DROP SUBSCRIPTION IF EXISTS r04_sub")
        primary.execute("DROP PUBLICATION IF EXISTS r04_pub")
        for conn in (primary, subscriber):
            conn.execute("DROP TABLE IF EXISTS orders")
            conn.execute("CREATE TABLE orders (id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, value text NOT NULL)")
        primary.execute("INSERT INTO orders(value) VALUES ('initial')")
        primary.execute("CREATE PUBLICATION r04_pub FOR TABLE orders")
        subscriber.execute("CREATE SUBSCRIPTION r04_sub CONNECTION 'host=primary port=5432 dbname=r04 user=r04 password=r04_local_only' PUBLICATION r04_pub")
        wait_for(lambda: subscriber.execute("SELECT id,value FROM orders ORDER BY id").fetchall() == [(1, "initial")], "initial table copy")


def experiment() -> dict[str, object]:
    setup()
    with connect("primary") as primary, connect("subscriber") as subscriber:
        primary.execute("DROP TABLE IF EXISTS unpublished")
        primary.execute("CREATE TABLE unpublished(id int)")
        primary.execute("INSERT INTO orders(value) VALUES ('inserted')")
        wait_for(lambda: subscriber.execute("SELECT id,value FROM orders ORDER BY id").fetchall() == [(1, "initial"), (2, "inserted")], "logical insert")
        primary.execute("UPDATE orders SET value='updated' WHERE id=2")
        wait_for(lambda: scalar(subscriber, "SELECT value FROM orders WHERE id=2") == "updated", "logical update")
        primary.execute("DELETE FROM orders WHERE id=1")
        wait_for(lambda: subscriber.execute("SELECT id,value FROM orders ORDER BY id").fetchall() == [(2, "updated")], "logical delete")
        assert scalar(subscriber, "SELECT to_regclass('public.unpublished')") is None
        return {"rows": [[2, "updated"]], "unpublished_absent": True, "operations": ["initial_copy", "insert", "update", "delete"]}


if __name__ == "__main__":
    print(json.dumps(experiment()))
```

```python
from __future__ import annotations

import json
import psycopg

from common import connect, scalar, wait_for
from v1 import setup


def experiment() -> dict[str, object]:
    setup()
    with connect("primary") as primary, connect("subscriber") as subscriber, connect("standby") as standby:
        primary.execute("ALTER TABLE orders ADD COLUMN note text")
        target = str(scalar(primary, "SELECT pg_current_wal_lsn()"))
        wait_for(lambda: standby.execute("SELECT pg_last_wal_replay_lsn() >= %s::pg_lsn", (target,)).fetchone()[0], "physical DDL")
        columns = "SELECT column_name FROM information_schema.columns WHERE table_name='orders' AND column_name='note'"
        assert primary.execute(columns).fetchall() == [("note",)]
        assert standby.execute(columns).fetchall() == [("note",)]
        assert subscriber.execute(columns).fetchall() == []
        subscriber.execute("ALTER TABLE orders ADD COLUMN note text")
        primary.execute("INSERT INTO orders(value,note) VALUES ('next','manual schema aligned')")
        wait_for(lambda: scalar(subscriber, "SELECT note FROM orders WHERE id=2") == "manual schema aligned", "aligned DDL data")
        try:
            subscriber.execute("INSERT INTO orders(value) VALUES ('local-write-collision')")
        except psycopg.errors.UniqueViolation:
            sequence_collision = True
        else:
            raise AssertionError("subscriber sequence unexpectedly followed publisher")
        primary_sequence = scalar(primary, "SELECT last_value FROM orders_id_seq")
        subscriber_sequence = scalar(subscriber, "SELECT last_value FROM orders_id_seq")
        assert primary_sequence == 2 and subscriber_sequence == 1
        primary.execute("ALTER TABLE orders REPLICA IDENTITY NOTHING")
        try:
            primary.execute("UPDATE orders SET value='must fail' WHERE id=2")
        except psycopg.errors.ObjectNotInPrerequisiteState:
            identity_rejected = True
        else:
            raise AssertionError("UPDATE accepted without replica identity")
        primary.execute("ALTER TABLE orders REPLICA IDENTITY FULL")
        primary.execute("UPDATE orders SET value='full identity' WHERE id=2")
        wait_for(lambda: scalar(subscriber, "SELECT value FROM orders WHERE id=2") == "full identity", "FULL identity update")
        primary.execute("DELETE FROM orders WHERE id=1")
        wait_for(lambda: scalar(subscriber, "SELECT count(*) FROM orders") == 1, "FULL identity delete")
        return {"physical_ddl": True, "logical_ddl_automatic": False, "sequence_collision": sequence_collision, "primary_sequence": primary_sequence, "subscriber_sequence": subscriber_sequence, "identity_rejected": identity_rejected, "full_identity_rows": [[2, "full identity"]]}


if __name__ == "__main__":
    print(json.dumps(experiment()))
```
