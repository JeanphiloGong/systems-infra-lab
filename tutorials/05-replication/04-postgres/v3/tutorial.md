# R04 v3：暂停消费者，观察槽保留 WAL

复制槽的 restart_lsn 不能越过暂停的逻辑消费者。v1 setup 后 DISABLE subscription，等待 pg_replication_slots.active=false，记录从当前 LSN 到 restart_lsn 的字节距离。primary 插入 20,000 行，再读取 retained_bytes；它必须大于暂停前的值。重新 ENABLE 后等待 subscriber 行数到 20,001，槽追平，保留距离下降到 0 或更小。

这不是 WAL 磁盘大小的完整测量：函数只测当前 LSN 与该槽 restart_lsn 的差，WAL 段、checkpoint、其他槽和归档会影响磁盘。暂停消费者长久运行会保留 WAL 并耗尽磁盘；恢复订阅只是让这个槽有机会前进，不保证已经被其他槽或归档需要的段立刻删除。

```python
from __future__ import annotations

import time
from collections.abc import Callable

import psycopg


def connect(node: str) -> psycopg.Connection:
    ports = {"primary": 55440, "standby": 55441, "subscriber": 55442}
    return psycopg.connect(host="127.0.0.1", port=ports[node], dbname="r04", user="r04", password="r04_local_only", autocommit=True, connect_timeout=3)


def wait_for(check: Callable[[], bool], label: str, seconds: float = 60) -> None:
    expires = time.monotonic() + seconds
    while time.monotonic() < expires:
        if check():
            return
        time.sleep(0.05)
    raise TimeoutError(label)


def scalar(conn: psycopg.Connection, query: str) -> object:
    row = conn.execute(query).fetchone()
    return None if row is None else row[0]
```

```python
from __future__ import annotations

import json

from common import connect, scalar, wait_for


def setup() -> None:
    with connect("subscriber") as subscriber, connect("primary") as primary:
        subscriber.execute("DROP SUBSCRIPTION IF EXISTS r04_sub")
        primary.execute("DROP PUBLICATION IF EXISTS r04_pub")
        for conn in (primary, subscriber):
            conn.execute("DROP TABLE IF EXISTS orders")
            conn.execute("CREATE TABLE orders (id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, value text NOT NULL)")
        primary.execute("INSERT INTO orders(value) VALUES ('initial')")
        primary.execute("CREATE PUBLICATION r04_pub FOR TABLE orders")
        subscriber.execute("CREATE SUBSCRIPTION r04_sub CONNECTION 'host=primary port=5432 dbname=r04 user=r04 password=r04_local_only' PUBLICATION r04_pub")
        wait_for(lambda: subscriber.execute("SELECT id,value FROM orders ORDER BY id").fetchall() == [(1, "initial")], "initial table copy")


def experiment() -> dict[str, object]:
    setup()
    with connect("primary") as primary, connect("subscriber") as subscriber:
        primary.execute("DROP TABLE IF EXISTS unpublished")
        primary.execute("CREATE TABLE unpublished(id int)")
        primary.execute("INSERT INTO orders(value) VALUES ('inserted')")
        wait_for(lambda: subscriber.execute("SELECT id,value FROM orders ORDER BY id").fetchall() == [(1, "initial"), (2, "inserted")], "logical insert")
        primary.execute("UPDATE orders SET value='updated' WHERE id=2")
        wait_for(lambda: scalar(subscriber, "SELECT value FROM orders WHERE id=2") == "updated", "logical update")
        primary.execute("DELETE FROM orders WHERE id=1")
        wait_for(lambda: subscriber.execute("SELECT id,value FROM orders ORDER BY id").fetchall() == [(2, "updated")], "logical delete")
        assert scalar(subscriber, "SELECT to_regclass('public.unpublished')") is None
        return {"rows": [[2, "updated"]], "unpublished_absent": True, "operations": ["initial_copy", "insert", "update", "delete"]}


if __name__ == "__main__":
    print(json.dumps(experiment()))
```

```python
from __future__ import annotations

import json

from common import connect, scalar, wait_for
from v1 import setup


def slot_bytes(conn) -> int:
    value = scalar(conn, "SELECT COALESCE(pg_wal_lsn_diff(pg_current_wal_lsn(), restart_lsn),0)::bigint FROM pg_replication_slots WHERE slot_name='r04_sub' AND NOT active")
    return int(value or 0)


def experiment() -> dict[str, object]:
    setup()
    with connect("primary") as primary, connect("subscriber") as subscriber:
        subscriber.execute("ALTER SUBSCRIPTION r04_sub DISABLE")
        wait_for(lambda: scalar(primary, "SELECT active IS FALSE FROM pg_replication_slots WHERE slot_name='r04_sub'") is True, "slot pause")
        before = slot_bytes(primary)
        primary.execute("INSERT INTO orders(value) SELECT 'paused-' || n FROM generate_series(1, 20000) AS n")
        after = slot_bytes(primary)
        assert after > before
        subscriber.execute("ALTER SUBSCRIPTION r04_sub ENABLE")
        wait_for(lambda: scalar(subscriber, "SELECT count(*) FROM orders") == 20001, "subscriber catch-up", seconds=90)
        wait_for(lambda: scalar(primary, "SELECT active IS TRUE FROM pg_replication_slots WHERE slot_name='r04_sub'") is True, "slot active")
        recovered = slot_bytes(primary)
        assert recovered < after
        return {"paused_bytes": before, "retained_bytes": after, "recovered_bytes": recovered, "subscriber_rows": 20001, "slot": "r04_sub"}


if __name__ == "__main__":
    print(json.dumps(experiment()))
```
